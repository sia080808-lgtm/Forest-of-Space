const {createCanvas,loadImage}=require('@napi-rs/canvas'),fs=require('node:fs'),path=require('node:path');
const dir=path.resolve(__dirname,'../app/src/main/assets/web/assets/forest/neolbi');
(async()=>{const im=await loadImage(dir+'/front.png'),src=createCanvas(im.width,im.height),sc=src.getContext('2d');sc.drawImage(im,0,0);const pixels=sc.getImageData(0,0,im.width,im.height).data,W=768,H=384,atlas=createCanvas(W*7,H*3),ac=atlas.getContext('2d');for(let frame=0;frame<21;frame++){const open=frame/20,c=createCanvas(W,H),ctx=c.getContext('2d'),out=ctx.createImageData(W,H);for(let x=0;x<W;x++){const dx=-23+(x+.5)*106/W;let lo=0,hi=1;for(let i=0;i<20;i++){const u=(lo+hi)/2,v=Math.max(0,(u-.70)/.30);if(-23+46*u+open*60*v*v<dx)lo=u;else hi=u;}const u=(lo+hi)/2,v=Math.max(0,(u-.70)/.30);if(dx>23+60*open)continue;const xx=85+1090*u;for(let y=0;y<H;y++){const dy=-58.5+(y+.5)*58.5/H,yy=40+(dy+35.5+open*16*Math.sin(v*Math.PI*.85))/27.2*645;if(yy<40||yy>=685)continue;const ix=Math.floor(xx),iy=Math.floor(yy),fx=xx-ix,fy=yy-iy,n=(y*W+x)*4;for(let k=0;k<4;k++)out.data[n+k]=pixels[(iy*im.width+ix)*4+k]*(1-fx)*(1-fy)+pixels[(iy*im.width+ix+1)*4+k]*fx*(1-fy)+pixels[((iy+1)*im.width+ix)*4+k]*(1-fx)*fy+pixels[((iy+1)*im.width+ix+1)*4+k]*fx*fy;}}ctx.putImageData(out,0,0);ac.drawImage(c,frame%7*W,Math.floor(frame/7)*H);}fs.writeFileSync(dir+'/canopy.png',atlas.toBuffer('image/png'));})();
